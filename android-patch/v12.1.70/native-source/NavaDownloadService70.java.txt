package com.verudanava.nava;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.os.Build;
import android.os.IBinder;
import java.util.concurrent.atomic.AtomicInteger;

public final class NavaDownloadService70 extends Service {
    private static final String CHANNEL_ID = "nava_downloads_v70";
    private static final int NOTIFICATION_ID = 70170;
    private static final String EXTRA_BATCH = "batch";
    private static final AtomicInteger ACTIVE = new AtomicInteger(0);

    public static void enqueue(Context context, String batch) {
        if (context == null || batch == null || batch.trim().isEmpty()) return;
        Context app = context.getApplicationContext();
        Intent i = new Intent(app, NavaDownloadService70.class);
        i.putExtra(EXTRA_BATCH, batch);
        if (Build.VERSION.SDK_INT >= 26) app.startForegroundService(i);
        else app.startService(i);
    }

    @Override public void onCreate() {
        super.onCreate();
        ensureChannel();
        startForeground(NOTIFICATION_ID, notification("Nava indiriyor", "İndirme arka planda devam ediyor"));
    }

    @Override public int onStartCommand(Intent intent, int flags, final int startId) {
        final String batch = intent == null ? null : intent.getStringExtra(EXTRA_BATCH);
        if (batch == null || batch.trim().isEmpty()) {
            stopSelfResult(startId);
            return START_NOT_STICKY;
        }
        ACTIVE.incrementAndGet();
        startForeground(NOTIFICATION_ID, notification("Nava indiriyor", "İndirme arka planda devam ediyor"));
        OfflineRuntime.submitBatch63(getApplicationContext(), batch, new Runnable() {
            @Override public void run() {
                if (ACTIVE.decrementAndGet() <= 0) {
                    ACTIVE.set(0);
                    try { stopForeground(true); } catch (Throwable ignored) {}
                    stopSelfResult(startId);
                }
            }
        });
        return START_REDELIVER_INTENT;
    }

    @Override public void onTaskRemoved(Intent rootIntent) {
        // Keep the foreground service alive when the UI task is removed.
        super.onTaskRemoved(rootIntent);
    }

    @Override public IBinder onBind(Intent intent) { return null; }

    private void ensureChannel() {
        if (Build.VERSION.SDK_INT < 26) return;
        NotificationManager nm = (NotificationManager) getSystemService(Context.NOTIFICATION_SERVICE);
        if (nm == null) return;
        NotificationChannel ch = new NotificationChannel(CHANNEL_ID, "Nava indirmeleri", NotificationManager.IMPORTANCE_LOW);
        ch.setDescription("Bölüm ve cilt indirmeleri arka planda devam eder");
        nm.createNotificationChannel(ch);
    }

    private Notification notification(String title, String text) {
        Notification.Builder b;
        if (Build.VERSION.SDK_INT >= 26) b = new Notification.Builder(this, CHANNEL_ID);
        else b = new Notification.Builder(this);
        return b.setSmallIcon(android.R.drawable.stat_sys_download)
                .setContentTitle(title)
                .setContentText(text)
                .setOngoing(true)
                .setOnlyAlertOnce(true)
                .setCategory(Notification.CATEGORY_PROGRESS)
                .build();
    }
}
